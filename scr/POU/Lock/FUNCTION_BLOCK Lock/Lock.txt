FUNCTION_BLOCK Lock
VAR_INPUT
	SysInfo	: sSystemInfo;
END_VAR
VAR_IN_OUT
	Data	: sDataLock;
	rData	: sRetainLock;
END_VAR
VAR
	i : UINT;
END_VAR
(*Синхронизация памяти*)
	SyncMemory();

FOR i := 1 TO 16 DO
	(*Обработка команд внешнего управления*)
		CASE Data.Item[i].Command OF
			eCommand.Enable	: Data.Item[i].RTD.RDD.enEnable	:= NOT Data.Item[i].RTD.RDD.enEnable; 
		END_CASE
		Data.Item[i].Command	:= eCommand.None;
	(*Конфигурирование таймера*)
		Data.Item[i].RTD.RDD.fbDelay (
								IN	:= Data.Item[i].Condition,
								PT	:= TO_TIME(Data.Item[i].RTD.TD*1000)
								);
	(*Основная логика*)
		(*Блокировка включена*)
			IF NOT Data.Item[i].RTD.RDD.enEnable THEN
				Data.Item[i].Trip	:= Data.Item[i].RTD.RDD.fbDelay.Q;
				(*Рассчитываем оставшееся время*)
					IF Data.Item[i].Condition AND NOT Data.Item[i].Trip THEN
						Data.Item[i].TimeLeft	:= TO_REAL(Data.Item[i].RTD.RDD.fbDelay.PT - Data.Item[i].RTD.RDD.fbDelay.ET) / 1000;
					ELSE
						Data.Item[i].TimeLeft	:= 0;
					END_IF
			
		(*Блокировка отключена*)
			ELSE
				Data.Item[i].Trip	:= FALSE;	//Сбрасываем сработку
			
				Data.Item[i].RTD.RDD.fbDelay	(IN	:= FALSE);	//Сбрасываем таймер
				Data.Item[i].TimeLeft			:= 0;			//Сбрасываем обратный отсчет
			END_IF
END_FOR

(*Агрегация*)
	(*Первопричина*)
		IF Data.Item[i].Trip THEN;
			Data.FirstIndex	:= TO_INT(i);
		END_IF
	(*Запрет активации*)
		Data.isLockActivate	:= FALSE;
		FOR i := 1 TO 8 DO
			IF Data.Item[i].Trip THEN
				Data.isLockActivate	:= TRUE;	
			END_IF
		END_FOR
	(*Запрет деактивации*)
		Data.isLockDeactivate	:= FALSE;
		FOR i := 9 TO 16 DO
			IF Data.Item[i].Trip THEN
				Data.isLockDeactivate	:= TRUE;	
			END_IF
		END_FOR
