FUNCTION_BLOCK Lock
VAR_INPUT
	SysInfo	: sSystemInfo;	//Информация о ПЛК
END_VAR
VAR_IN_OUT
	Data	: sDataLock;	//Контейнер данных
	rData	: sRetainLock;	//Контейнер retain-данных
END_VAR
VAR
	i			: UINT;	//Счетчик
	isCmdOn		: BOOL;	//Аргумент команды: TRUE - включить, FALSE - выключить
	firstTrip	: INT;	//Сработавший ключ с меньшим номером в текущем цикле
END_VAR
(*
	Набор технологических блокировок одного механизма.
	Ключи 1..8 - запрет активации (открытия / запуска), 9..16 - запрет деактивации (закрытия / остановки).
	Блокировка не самоудерживается: условие пропало - ключ отпустил.
	Первопричина фиксируется до команды Reset и на запреты не влияет.

	Правило: экземпляр ФБ не хранит состояния между вызовами, состояние - только в Data/rData.
	Внутри тела нет RETURN: SyncOut обязан выполняться в каждом цикле.
*)

(*Синхронизация памяти - чтение*)
	SyncIn();

(*Обработка команд внешнего управления.
SCADA пишет CmdValue и CmdIndex, затем Command. Ведомый команды не исполняет*)
	IF NOT SysInfo.RDD_Standby AND Data.Command <> eCommand.None THEN
		isCmdOn	:= Data.CmdValue <> 0.0;
		CASE Data.Command OF
			eCommand.Reset:
				IF Data.CmdIndex = 0 THEN
					Data.RDD.FirstIndex	:= 0;
				END_IF
			eCommand.Enable:
				IF Data.CmdIndex >= 1 AND Data.CmdIndex <= 16 THEN
					Data.Item[Data.CmdIndex].RTD.RDD.enEnable	:= isCmdOn;
				END_IF
		END_CASE
	(*Логирование: сдвиг истории от новых к старым, [1] - последняя команда*)
		FOR i := 5 TO 2 BY -1 DO
			Data.RDD.Logger[i]	:= Data.RDD.Logger[i-1];
		END_FOR
		Data.RDD.Logger[1].Command	:= Data.Command;
		Data.RDD.Logger[1].Source	:= Data.Source;
		Data.RDD.Logger[1].Value	:= Data.CmdValue;
		Data.RDD.Logger[1].Index	:= Data.CmdIndex;
	END_IF
(*Сброс интерфейса управления*)
	Data.Command	:= eCommand.None;
	Data.Source		:= eCommandSource.None;
	Data.CmdIndex	:= 0;

(*Обработка ключей*)
	Data.isLockActivate		:= FALSE;
	Data.isLockDeactivate	:= FALSE;
	firstTrip				:= 0;
	FOR i := 1 TO 16 DO
	(*Сработка - по истечении задержки, только у ключа в работе*)
		Data.Item[i].Trip	:= fTimerOn(Tmr			:= Data.Item[i].RTD.RDD.Delay,
										IN			:= Data.Item[i].Condition AND Data.Item[i].RTD.RDD.enEnable,
										PT			:= Data.Item[i].RTD.TD,
										CycleTime	:= SysInfo.CycleTime);
	(*Обратный отсчет до сработки*)
		IF Data.Item[i].RTD.RDD.Delay.State = eTimerState.Running THEN
			Data.Item[i].TimeLeft	:= Data.Item[i].RTD.TD - Data.Item[i].RTD.RDD.Delay.ET / 1000;
		ELSE
			Data.Item[i].TimeLeft	:= 0.0;
		END_IF
	(*Агрегация запретов*)
		IF Data.Item[i].Trip THEN
			IF i <= 8 THEN
				Data.isLockActivate		:= TRUE;
			ELSE
				Data.isLockDeactivate	:= TRUE;
			END_IF
			IF firstTrip = 0 THEN
				firstTrip	:= TO_INT(i);
			END_IF
		END_IF
	END_FOR

(*Первопричина: фиксируется первый сработавший ключ (при одновременной сработке - меньший номер).
Хранится только в RDD.FirstIndex (retain), для SCADA - IO.Lock[n].RDD.FirstIndex*)
	IF Data.RDD.FirstIndex = 0 THEN
		Data.RDD.FirstIndex	:= firstTrip;
	END_IF

(*Синхронизация памяти - запись*)
	SyncOut();
