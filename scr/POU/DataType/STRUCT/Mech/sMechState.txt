TYPE sMechState :
STRUCT
(*Слово 1. Режимы и состояния*)
	(*Байт 0. Режимы управления*)
		Auto		: BIT;	//Бит 0 - Автоматический режим
		Manual		: BIT;	//Бит 1 - Ручной режим
		Local		: BIT;	//Бит 2 - Местный
		Remote		: BIT;	//Бит 3 - Дистанционный
		KeyVirtual	: BIT;	//Бит 4 - Тип ключа
		Repair		: BIT;	//Бит 5 - Ремонт
		ReservMode6	: BIT;	//Бит 6 - Резерв
		ReservMode7	: BIT;	//Бит 7 - Резерв
	(*Байт 1. Текущие состояния*)
		Opened			: BIT;	//Бит 0 - Открыт
		Closed			: BIT;	//Бит 1 - Закрыт
		Running			: BIT;	//Бит 2 - Работает
		Stopped			: BIT;	//Бит 3 - Остановлен
		Opening			: BIT;	//Бит 4 - Открывается
		Closing			: BIT;	//Бит 5 - Закрывается
		Intermediate	: BIT;	//Бит 6 - Промежуточное положение
		Ready			: BIT;	//Бит 7 - Готовность
(*Слово 2. Аварии и Блокировки*)
	(*Байт 2. Аварии хода и оборудования*)
		Alarm			: BIT;	//Бит 0 - Обобщенная авария
		Unknown			: BIT;	//Бит 1 - Неопределенное состяоние
		EStop			: BIT;	//Бит 2 - Нажата кнопка "Аварийный останов"
		ErrStartOpen	: BIT;	//Бит 3 - Нет движения на открытие / запуск
		ErrStartClose	: BIT;	//Бит 4 - Нет движения на закрытие / остановку
		ErrFullOpen		: BIT;	//Бит 5 - Таймаут открытия / пуска
		ErrFullClose	: BIT;	//Бит 6 - Таймаут закрытия /остановки
		ErrUncmdStart	: BIT;	//Бит 7 - Несанкционированное открытие / запуск
	(*Байт 3. Прочие аварии и Блокировки*)
		ErrUncmdStop	: BIT;	//Бит 0 - Несанкционированное закрытие / отключение
		ErrSensors		: BIT;	//Бит 1 - Недостоверность датчиков / концевиков
		ErrVFDLink		: BIT;	//Бит 2 - Потеря связи с ПЧ
		lkStart			: BIT;	//Бит 3 - Блокировка пуска
		lkStop			: BIT;	//Бит 4 - Блокировка останова
		lkOpen			: BIT;	//Бит 5 - Блокировка открытия
		lkClose			: BIT;	//Бит 6 - Блокировка закрытия
		EPSActive		: BIT;	//Бит 7 - Сработка ПАЗ 
END_STRUCT
END_TYPE
