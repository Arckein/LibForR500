{attribute 'qualified_only'}
{attribute 'strict'}
TYPE sSignalState :
STRUCT
(*Слово 1. Режимы и диагностика*)
	(*Байт 1. Режимы работы сигнала*)
		Disabled	: BIT;	//Бит 0	- Признак режима "Обработка логики отключена"
		Imitation	: BIT;	//Бит 1	- Признак режима "Имитация"
		Freezing	: BIT;	//Бит 2	- Признак режима "Заморозка последнего значения"
		Filter		: BIT;	//Бит 3	- Признак режима "Фильтрация"
		Invert		: BIT;	//Бит 4	- Признак режима "Инверсия"
		CutOff		: BIT;	//Бит 5	- Признак режима "Мертвая зона"
		AlarmMode	: BIT;	//Бит 6	- Признак режима "Сигнализация аварий"
		Repair		: BIT;	//Бит 7	- Признак режима "Ремонт"
	(*Байт 2. Ошибки*)
		ModuleFault		: BIT;	//Бит 0 - Ошибка "Неисправность модуля"
		ChFault			: BIT;	//Бит 1 - Ошибка "Неисправность канала"
		ExternalBreak	: BIT;	//Бит 2 - Ошибка "Обрыв цепи"
		ConfigFault		: BIT;	//Бит 3 - Ошибка "Ошибка конфигурации сигнала"
		ShortCircuit	: BIT;	//Бит 4 - Ошибка "Выше порога достоверности"
		Break			: BIT;	//Бит 5 - Ошибка "Ниже порога достоверности"
		Reserv14		: BIT;	//Бит 6 - Резерв
		Reserv15		: BIT;	//Бит 7 - Резерв
		
(*Слово 2. Сигнализации*)
	(*Байт 3-4*)
		ActiveAlarm	:	eAlarmType;
END_STRUCT
END_TYPE
