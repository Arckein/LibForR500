(*
	Таймер задержки включения библиотеки (замена TON).
	Собственной памяти нет: все состояние - в Tmr (sTimer в retain-части объекта).
	Вызывается КАЖДЫЙ цикл с фактическим входом IN.

	Машина состояний:
		Ready	--IN-->					Running	(первый цикл отсчета)
		Running	--ET >= PT-->			Done
		Done	--PT увеличили-->		Running	(отсчет продолжается)
		любое	--NOT IN-->				Ready	(ET = 0)
		Reset	-->						Ready	(ET = 0), при IN = TRUE в том же вызове - старт отсчета
	Сброс при активном входе: владелец пишет Tmr.State := eTimerState.Reset.

	PT - секунды (как TD/TF в retain-данных), CycleTime - мс (sSystemInfo.CycleTime).
	PT <= 0 - Done в первом же цикле с IN = TRUE.
*)
FUNCTION fTimerOn : BOOL
VAR_IN_OUT
	Tmr	: sTimer;	//Таймер (в retain-части объекта)
END_VAR
VAR_INPUT
	IN			: BOOL;	//Вход
	PT			: REAL;	//Уставка, сек
	CycleTime	: REAL;	//Время цикла задачи, мс
END_VAR
Tmr.PT	:= MAX(PT * 1000, 0.0);

(*Сброс: по запросу владельца или при снятии входа*)
	IF Tmr.State = eTimerState.Reset OR NOT IN THEN
		Tmr.ET		:= 0.0;
		Tmr.State	:= eTimerState.Ready;
	END_IF

(*Машина состояний*)
	CASE Tmr.State OF
		eTimerState.Ready:
		(*Готов к работе: старт по входу, первый цикл отсчета*)
			IF IN THEN
				Tmr.ET		:= CycleTime;
				Tmr.State	:= eTimerState.Running;
			END_IF

		eTimerState.Running:
		(*В работе: отсчет*)
			IF Tmr.ET < Tmr.PT THEN
				Tmr.ET	:= Tmr.ET + CycleTime;
			END_IF

		eTimerState.Done:
		(*Закончил отсчет: ждет снятия входа или сброса.
		Уставку увеличили на ходу - продолжает отсчет*)
			IF Tmr.ET < Tmr.PT THEN
				Tmr.ET		:= Tmr.ET + CycleTime;
				Tmr.State	:= eTimerState.Running;
			END_IF
	ELSE
	(*Недопустимое значение (например, испорченная retain-память) - в исходное состояние*)
		Tmr.ET		:= 0.0;
		Tmr.State	:= eTimerState.Ready;
	END_CASE

(*Окончание отсчета*)
	IF Tmr.State = eTimerState.Running AND Tmr.ET >= Tmr.PT THEN
		Tmr.ET		:= Tmr.PT;	//Отсчет не растет дальше уставки
		Tmr.State	:= eTimerState.Done;
	END_IF

fTimerOn	:= Tmr.State = eTimerState.Done;
