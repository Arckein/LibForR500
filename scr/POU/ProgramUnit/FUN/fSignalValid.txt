(*
	Единое правило достоверности сигнала для потребителей (механизмы, блокировки, ПАЗ).
	Потребитель НЕ анализирует Errors напрямую - только через эту функцию.

	Отключен (резерв)	-> недостоверен
	Имитация			-> достоверен (значение подменено осознанно, ошибки канала видны оператору)
	Работа				-> достоверен при отсутствии ошибок
*)
FUNCTION fSignalValid : BOOL
VAR_INPUT
	Status	: uSignalStatus;	//Комплексный статус сигнала
END_VAR
fSignalValid	:= NOT Status.State.Disabled
					AND (Status.State.Imitation OR Status.MemoryBranch.Errors = 0);
