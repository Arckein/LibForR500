(*
	Функция получает текущее время задачи, в которой находится POU по ее номеру.
	Для корректной работы функции в Менеджер библиотек необходимо добавлять библиотеку CmpIecTask.
	На вход функции (переменная Х) необходимо вписать номер задачи от 0 до N, где N - последняя по счету задача в конфигураторе задач.
	Задачи нумеруются сверху вниз в соответствии с конфигуратором задач в дереве устройств ПЛК.
*)
FUNCTION fCycleTime : REAL
VAR
	HandleTask	: CmpIecTask.RTS_IEC_HANDLE;		//Handle задачи, которая подана на вход функции
	TaskInfo	: POINTER TO PsTechEE.Task_Info2;	//Информация о задаче, которая подана на вход функции
	SysResult   : CmpIecTask.RTS_IEC_RESULT;        // Заглушка для системного ответа
END_VAR
