FUNCTION_BLOCK AnalogIn
VAR_INPUT
	SysInfo	: sSystemInfo;	//Информация о ПЛК
END_VAR
VAR_IN_OUT
	Data	: sDataAI;		//Контейнер данных
	rData	: sRetainAI;	//Контейнер retain-данных
END_VAR
VAR
	localIn		: REAL;	//Буферизированное значение "Вход с канала"
	i			: UINT;	//Счетчик
	isValidIn	: BOOL;	//Канал исправен в текущем цикле
	isHold		: BOOL;	//Значение берется из буфера последнего достоверного (Заморозка)
	isCmdOn		: BOOL;	//Аргумент команды: TRUE - включить, FALSE - выключить
END_VAR
(*
	Правило: экземпляр ФБ не хранит состояния между вызовами.
	Все переменные VAR пересчитываются до чтения, состояние - только в Data/rData.
	Поэтому один экземпляр обслуживает весь массив AI.
	Внутри тела нет RETURN: SyncOut обязан выполняться в каждом цикле.
*)

(*Синхронизация памяти - чтение*)
	SyncIn();

(*Буферизация входных данных*)
	localIn			:= Data.In;
	Data.RawValue	:= localIn;

(*Обработка команд внешнего управления.
Команда + аргумент: повторная запись дает тот же результат.
SCADA пишет сначала CmdValue, затем Command.
Ведомый команды не исполняет - состояние придет с ведущего*)
	IF NOT SysInfo.RDD_Standby AND Data.Command <> eCommand.None THEN
		isCmdOn	:= Data.CmdValue <> 0.0;
		CASE Data.Command OF
			eCommand.Enable:
				Data.RTD.RDD.enEnable		:= isCmdOn;
			eCommand.Imit:
				Data.RTD.RDD.enImit			:= isCmdOn;
			eCommand.Freezing:
				Data.RTD.RDD.enFreezing		:= isCmdOn;
			eCommand.Filter:
				Data.RTD.RDD.enFilter		:= isCmdOn;
			eCommand.AlarmMode:
				Data.RTD.RDD.enAlarmMode	:= isCmdOn;
			eCommand.UseHH:
				Data.RTD.RDD.enUseHH		:= isCmdOn;
			eCommand.UseH:
				Data.RTD.RDD.enUseH			:= isCmdOn;
			eCommand.UseL:
				Data.RTD.RDD.enUseL			:= isCmdOn;
			eCommand.UseLL:
				Data.RTD.RDD.enUseLL		:= isCmdOn;
			eCommand.Delay:
				Data.RTD.RDD.enDelay		:= isCmdOn;
			eCommand.CutOff:
				Data.RTD.RDD.enCutOff		:= isCmdOn;
		END_CASE
	(*Логирование: сдвиг истории от новых к старым, [1] - последняя команда*)
		FOR i := 5 TO 2 BY -1 DO
			Data.RTD.RDD.Logger[i]	:= Data.RTD.RDD.Logger[i-1];
		END_FOR
		Data.RTD.RDD.Logger[1].Command	:= Data.Command;
		Data.RTD.RDD.Logger[1].Source	:= Data.Source;
		Data.RTD.RDD.Logger[1].Value	:= Data.CmdValue;
	END_IF
(*Сброс интерфейса управления*)
	Data.Command	:= eCommand.None;
	Data.Source		:= eCommandSource.None;

(*Основная логика - выкл (резервный канал: обработка не выполняется)*)
	IF NOT Data.RTD.RDD.enEnable THEN
		Data.EngValue					:= 0.0;
		Data.RTD.RDD.isFirstPass		:= TRUE;
		Data.RTD.RDD.alarmTimer			:= 0.0;
		Data.Status.MemoryBranch.Errors	:= 0;
		Data.Status.State.ActiveAlarm	:= eAlarmType.No;

(*Основная логика - вкл*)
	ELSE
	(*Валидация. Ошибки канала показываются ВСЕГДА, в т.ч. в имитации*)
		Validation(Data);
		isValidIn	:= Data.Status.MemoryBranch.Errors = 0;

	(*Предобработка исправного канала: фильтр и буфер последнего достоверного*)
		IF isValidIn THEN
			IF Data.RTD.RDD.enFilter THEN
				localIn	:= ApplyFilter(Data);
			ELSE
				Data.RTD.RDD.isFirstPass	:= TRUE;
			END_IF
			Data.RTD.RDD.freezingIn	:= localIn;
		ELSE
			(*После восстановления канала фильтр стартует с актуального значения*)
			Data.RTD.RDD.isFirstPass	:= TRUE;
		END_IF

	(*Формирование значения*)
		IF Data.RTD.RDD.enImit THEN
		(*Имитация: значение задает оператор*)
			Data.EngValue	:= Data.RTD.RDD.ImitVal;
		ELSE
		(*Ошибка канала с включенной "Заморозкой" - берем буфер последнего достоверного*)
			isHold	:= NOT isValidIn AND Data.RTD.RDD.enFreezing;

			IF Data.Status.State.ConfigFault OR (NOT isValidIn AND NOT isHold) THEN
				Data.EngValue	:= Data.RTD.YMin;
			ELSE
				IF isHold THEN
					localIn	:= Data.RTD.RDD.freezingIn;
				END_IF
				Data.EngValue	:= ScaleValue(Data);
			(*Мертвая зона*)
				IF Data.RTD.RDD.enCutOff THEN
					IF ABS(Data.EngValue - Data.RTD.YMin) < Data.RTD.Threshold THEN
						Data.EngValue	:= Data.RTD.YMin;
					END_IF
				END_IF
			(*Значение для безударного перехода в Имитацию*)
				Data.RTD.RDD.ImitVal	:= Data.EngValue;
			END_IF
		END_IF

	(*Сигнализации: в имитации - всегда (проверка логики), в работе - только по исправному каналу*)
		IF Data.RTD.RDD.enAlarmMode AND (Data.RTD.RDD.enImit OR isValidIn) THEN
			Data.Status.State.ActiveAlarm	:= CheckAlarms(Data);
		ELSE
			Data.Status.State.ActiveAlarm	:= eAlarmType.No;
			Data.RTD.RDD.alarmTimer			:= 0.0;
		END_IF
	END_IF

(*Упаковка состояний режимов - после обработки, чтобы статус отражал итог цикла*)
	Data.Status.State.Disabled	:= NOT Data.RTD.RDD.enEnable;
	Data.Status.State.Imitation	:= Data.RTD.RDD.enImit;
	Data.Status.State.Freezing	:= Data.RTD.RDD.enFreezing;
	Data.Status.State.Filter	:= Data.RTD.RDD.enFilter;
	Data.Status.State.CutOff	:= Data.RTD.RDD.enCutOff;
	Data.Status.State.AlarmMode	:= Data.RTD.RDD.enAlarmMode;

(*Синхронизация памяти - запись*)
	SyncOut();
