TYPE sDataDO :
STRUCT
(*Входные значения*)
	EngValue		: BOOL;			//Команда от логики управления
	ChStatus		: BOOL;			//Ошибка канала (TRUE - Ошибка)
	ModuleStatus	: BOOL;			//Ошибка модуля (TRUE - Ошибка)
	Command			: eCommand;		//Команды внешнего управления
	CmdValue		: REAL;			//Аргумент команды: 0 - выкл, <>0 - вкл (писать ДО Command)
	Source			: eCommandSource;	//Источник команды
	RTD				: sRetainDO;	//Хранимые данные
    
(*Выходные значения*)
	Out		: BOOL;				//Итоговый сырой сигнал на физический канал
	Status	: uSignalStatus;	//Текущий статус достоверности
END_STRUCT
END_TYPE
