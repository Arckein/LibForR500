METHOD ApplyFilter : BOOL
VAR_IN_OUT
	mData	: sDataDI;
END_VAR
(*Старт фильтра: пуск ПЛК или восстановление канала после ошибки.
Принимаем текущее значение без задержки*)
	IF SysInfo.isInit OR SysInfo.isColdStart OR mData.RTD.RDD.isFirstPass THEN
		mData.RTD.RDD.filteredIn	:= localIn;
		mData.RTD.RDD.isFirstPass	:= FALSE;
		fTimerOn(Tmr := mData.RTD.RDD.filterTimer, IN := FALSE, PT := 0.0, CycleTime := 0.0);	//Сброс таймера
		ApplyFilter					:= localIn;
		RETURN;
	END_IF
(*Защита от нулевого времени фильтра*)
	IF mData.RTD.TF <= 0.001 THEN
		mData.RTD.RDD.filteredIn	:= localIn;
		fTimerOn(Tmr := mData.RTD.RDD.filterTimer, IN := FALSE, PT := 0.0, CycleTime := 0.0);	//Сброс таймера
		ApplyFilter					:= localIn;
		RETURN;
	END_IF

(*Антидребезг: новое значение принимается, если оно держится не меньше TF.
Таймер идет, пока вход отличается от отфильтрованного значения, и сбрасывается при совпадении*)
	IF fTimerOn(Tmr			:= mData.RTD.RDD.filterTimer,
				IN			:= localIn <> mData.RTD.RDD.filteredIn,
				PT			:= mData.RTD.TF,
				CycleTime	:= SysInfo.CycleTime) THEN
		mData.RTD.RDD.filteredIn	:= localIn;
		fTimerOn(Tmr := mData.RTD.RDD.filterTimer, IN := FALSE, PT := 0.0, CycleTime := 0.0);	//Сброс таймера
	END_IF
ApplyFilter	:= mData.RTD.RDD.filteredIn;
