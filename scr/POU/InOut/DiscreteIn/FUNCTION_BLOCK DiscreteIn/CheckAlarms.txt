METHOD CheckAlarms : eAlarmType
VAR
	isAlarmLevel	: BOOL;	//Значение в аварийном состоянии
	isDelayDone		: BOOL;	//Задержка сигнализации отсчитана
END_VAR
VAR_IN_OUT
	mData	: sDataDI;
END_VAR
isAlarmLevel	:= mData.EngValue = mData.RTD.AlarmLevel;
isDelayDone		:= fTimerOn(Tmr			:= mData.RTD.RDD.alarmTimer,
							IN			:= isAlarmLevel AND mData.RTD.RDD.enDelay,
							PT			:= mData.RTD.TD,
							CycleTime	:= SysInfo.CycleTime);

IF isAlarmLevel AND (NOT mData.RTD.RDD.enDelay OR isDelayDone) THEN
	CheckAlarms	:= eAlarmType.Yes;
ELSE
	CheckAlarms	:= eAlarmType.No;
END_IF
